@st.cache_resource
def connect_database(uri):
    client = MongoClient(uri, serverSelectionTimeoutMS=5000)
    client.admin.command("ping")

    db = client["blog_website"]
    users = db["users"]
    blogs = db["blogs"]

    users.create_index("username", unique=True)
    users.create_index("email", unique=True)
    blogs.create_index([("created_at", -1)])

    return users, blogs


if not MONGO_URI:
    st.error("MongoDB connection string is missing. Add MONGO_URI to your .env file.")
    st.stop()

    users, blogs = connect_database(MONGO_URI)
except Exception as error:

# ---------------- HELPER FUNCTIONS ----------------

def register_user(username, email, password):
    username = username.strip()
    email = email.strip().lower()

    if not username or not email or not password:
        return False, "Please fill all fields."

    if len(password) < 8:
        return False, "Password must contain at least 8 characters."

    hashed_password = bcrypt.hashpw(
        password.encode("utf-8"),
        bcrypt.gensalt()
    )

    try:
        users.insert_one({
            "username": username,
            "email": email,
            "password": hashed_password,
            "created_at": datetime.now(timezone.utc)
        })
        return True, "Registration successful!"

    except Exception:
        return False, "Username or email already exists."


def login_user(username, password):
    user = users.find_one({
        "username": username.strip()
    })

    if user and bcrypt.checkpw(
        password.encode("utf-8"),
        user["password"]
    ):
        return user

    return None


def create_blog(title, category, image, description, author):
    blogs.insert_one({
        "title": title.strip(),
        "category": category,
        "image": image.strip(),
        "description": description.strip(),
        "author": author,
        "created_at": datetime.now(timezone.utc)
    })


def get_all_blogs(search=""):
    query = {}

    if search.strip():
        query["$or"] = [
            {"title": {"$regex": search, "$options": "i"}},
            {"category": {"$regex": search, "$options": "i"}}
        ]

    return list(
        blogs.find(query).sort("created_at", -1)
    )


def delete_blog(blog_id, username):
    blogs.delete_one({
        "_id": ObjectId(blog_id),
        "author": username
    })


def update_blog(blog_id, title, category, image, description, username):
    blogs.update_one(
        {
            "_id": ObjectId(blog_id),
            "author": username
        },
        {
            "$set": {
                "title": title.strip(),
                "category": category,
                "image": image.strip(),
                "description": description.strip()
            }
        }
    )
